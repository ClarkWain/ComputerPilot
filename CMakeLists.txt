cmake_minimum_required(VERSION 3.29)
project(DesktopPilot VERSION 0.4.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This first backend requires Windows (WIC image I/O).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "LibTorch package ABI" FORCE)
option(DP_WITH_TORCH "Build the real OmniParser TorchScript detector" ON)
option(DP_WITH_OCR "Build native PP-OCRv4 via ONNX Runtime" OFF)
option(DP_WITH_WEBP "Encode agent screenshots with lightweight libwebp" ON)
set(DP_ORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.deps/onnxruntime-win-x64-1.23.2" CACHE PATH "Native ONNX Runtime")
set(DP_TEST_OCR "" CACHE PATH "Optional OCR model directory for inference CTest")
set(DP_TEST_MODEL "" CACHE FILEPATH "Optional real model for inference CTest")

add_library(dp_core src/core.cpp src/image_windows.cpp src/ocr_common.cpp src/preflight.cpp)
target_include_directories(dp_core PUBLIC include third_party)
target_compile_definitions(dp_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(dp_core PUBLIC /utf-8 /EHsc PRIVATE /W4)
target_link_libraries(dp_core PRIVATE windowscodecs ole32 gdi32 bcrypt)
if(DP_WITH_WEBP)
  include(FetchContent)
  foreach(component ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS)
    set(WEBP_BUILD_${component} OFF CACHE BOOL "Only build the WebP codec" FORCE)
  endforeach()
  set(WEBP_LINK_STATIC ON CACHE BOOL "Static WebP codec" FORCE)
  set(webp_archive "https://storage.googleapis.com/downloads.webmproject.org/releases/webp/libwebp-1.6.0.tar.gz")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/downloads/libwebp-1.6.0.tar.gz")
    set(webp_archive "${CMAKE_CURRENT_SOURCE_DIR}/.deps/downloads/libwebp-1.6.0.tar.gz")
  endif()
  FetchContent_Declare(webp URL "${webp_archive}"
    URL_HASH SHA256=e4ab7009bf0629fd11982d4c2aa83964cf244cffba7347ecd39019a9e38c4564
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(webp)
  target_link_libraries(dp_core PRIVATE webp)
  target_compile_definitions(dp_core PRIVATE DP_WITH_WEBP=1)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/licenses/webp")
  configure_file("${webp_SOURCE_DIR}/COPYING" "${CMAKE_BINARY_DIR}/licenses/webp/COPYING" COPYONLY)
  configure_file("${webp_SOURCE_DIR}/PATENTS" "${CMAKE_BINARY_DIR}/licenses/webp/PATENTS" COPYONLY)
endif()

if(DP_WITH_TORCH)
  find_package(Torch REQUIRED)
  add_library(dp_detector src/detector_torch.cpp)
  target_link_libraries(dp_detector PUBLIC dp_core PRIVATE "${TORCH_LIBRARIES}")
  target_compile_options(dp_detector PRIVATE /bigobj /W4)
  target_compile_definitions(dp_detector PUBLIC DP_WITH_TORCH=1)
else()
  add_library(dp_detector src/detector_unavailable.cpp)
  target_link_libraries(dp_detector PUBLIC dp_core)
endif()

if(DP_WITH_OCR)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
  if(NOT EXISTS "${DP_ORT_ROOT}/include/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "ONNX Runtime missing. Run cmake -P cmake/bootstrap-ocr.cmake")
  endif()
  add_library(dp_ort SHARED IMPORTED)
  set_target_properties(dp_ort PROPERTIES
    IMPORTED_LOCATION "${DP_ORT_ROOT}/lib/onnxruntime.dll"
    IMPORTED_IMPLIB "${DP_ORT_ROOT}/lib/onnxruntime.lib"
    INTERFACE_INCLUDE_DIRECTORIES "${DP_ORT_ROOT}/include")
  add_library(dp_ocr src/ocr_onnx.cpp)
  target_link_libraries(dp_ocr PUBLIC dp_core PRIVATE dp_ort opencv_core opencv_imgproc)
  target_compile_options(dp_ocr PRIVATE /W4)
else()
  add_library(dp_ocr src/ocr_unavailable.cpp)
  target_link_libraries(dp_ocr PUBLIC dp_core)
endif()

add_library(dp_desktop src/desktop_windows.cpp)
target_link_libraries(dp_desktop PUBLIC dp_core PRIVATE uiautomationcore oleaut32 user32 wtsapi32)
add_library(dp_service src/service.cpp src/semantic.cpp src/semantic_actions.cpp src/semantic_modern_paint.cpp src/agent_api.cpp src/profiles.cpp)
target_link_libraries(dp_service PUBLIC dp_detector dp_ocr dp_desktop)
add_executable(desktop-pilot src/main.cpp src/server.cpp src/worker_windows.cpp)
target_link_libraries(desktop-pilot PRIVATE dp_service)
add_executable(desktop-pilot-onboarding WIN32 examples/onboarding/fixture.cpp)
target_compile_definitions(desktop-pilot-onboarding PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(desktop-pilot-onboarding PRIVATE user32 gdi32)
if(DP_WITH_OCR)
  add_custom_command(TARGET desktop-pilot POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "$<TARGET_FILE:dp_ort>" "$<TARGET_FILE:opencv_core>" "$<TARGET_FILE:opencv_imgproc>" "$<TARGET_FILE_DIR:desktop-pilot>")
endif()
if(DP_WITH_TORCH)
  file(GLOB TORCH_DLLS "${TORCH_INSTALL_PREFIX}/lib/*.dll")
  add_custom_command(TARGET desktop-pilot POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${TORCH_DLLS} $<TARGET_FILE_DIR:desktop-pilot>
    COMMAND_EXPAND_LISTS)
endif()
