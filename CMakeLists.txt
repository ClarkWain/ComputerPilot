cmake_minimum_required(VERSION 3.29)
project(ComputerPilot VERSION 0.6.1 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS version")
elseif(NOT WIN32)
  message(FATAL_ERROR "Supported desktop backends: Windows and macOS")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

  set(DP_TORCH_DEFAULT ON)

  set(DP_TORCH_DEFAULT OFF)
endif()
option(DP_WITH_TORCH "Build the real OmniParser TorchScript detector" ${DP_TORCH_DEFAULT})
option(DP_WITH_OCR "Build native PP-OCRv4 via ONNX Runtime" OFF)
option(DP_WITH_WEBP "Encode agent screenshots with lightweight libwebp" ON)
option(DP_STATIC_RUNTIME "Statically link the MSVC runtime for model-free Core" OFF)
if(DP_STATIC_RUNTIME AND (DP_WITH_TORCH OR DP_WITH_OCR))
  message(FATAL_ERROR "DP_STATIC_RUNTIME requires DP_WITH_TORCH=OFF and DP_WITH_OCR=OFF")
endif()
if(DP_STATIC_RUNTIME)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")

  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")

if(MSVC)
  target_compile_definitions(dp_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_compile_options(dp_core PUBLIC /utf-8 /EHsc PRIVATE /W4)

  target_compile_options(dp_core PRIVATE -Wall -Wextra)
endif()

  target_link_libraries(dp_core PRIVATE windowscodecs ole32 gdi32 bcrypt)
else()
  target_compile_options(dp_core PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(dp_core PRIVATE "-framework AppKit" "-framework ImageIO")
endif()
if(DP_WITH_WEBP)
  include(FetchContent)
  foreach(component ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS)
    set(WEBP_BUILD_${component} OFF CACHE BOOL "Only build the WebP codec" FORCE)
  endforeach()
  set(WEBP_LINK_STATIC ON CACHE BOOL "Static WebP codec" FORCE)
  set(webp_archive "https://storage.googleapis.com/downloads.webmproject.org/releases/webp/libwebp-1.6.0.tar.gz")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/downloads/libwebp-1.6.0.tar.gz")
    set(webp_archive "${CMAKE_CURRENT_SOURCE_DIR}/.deps/downloads/libwebp-1.6.0.tar.gz")
  endif()
  FetchContent_Declare(webp URL "${webp_archive}"
    URL_HASH SHA256=e4ab7009bf0629fd11982d4c2aa83964cf244cffba7347ecd39019a9e38c4564
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(webp)
  target_link_libraries(dp_core PRIVATE webp)
  target_compile_definitions(dp_core PRIVATE DP_WITH_WEBP=1)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/licenses/webp")
  configure_file("${webp_SOURCE_DIR}/COPYING" "${CMAKE_BINARY_DIR}/licenses/webp/COPYING" COPYONLY)
  configure_file("${webp_SOURCE_DIR}/PATENTS" "${CMAKE_BINARY_DIR}/licenses/webp/PATENTS" COPYONLY)
endif()

if(DP_WITH_TORCH)
  find_package(Torch REQUIRED)
  add_library(dp_detector src/detector_torch.cpp)
  target_link_libraries(dp_detector PUBLIC dp_core PRIVATE "${TORCH_LIBRARIES}")
  if(MSVC)
    target_compile_options(dp_detector PRIVATE /bigobj /W4)
  endif()
  target_compile_definitions(dp_detector PUBLIC DP_WITH_TORCH=1)

  add_library(dp_detector src/detector_unavailable.cpp)
  target_link_libraries(dp_detector PUBLIC dp_core)
endif()

if(DP_WITH_OCR)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
  if(NOT EXISTS "${DP_ORT_ROOT}/include/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "ONNX Runtime missing. Set DP_ORT_ROOT to the native SDK; Windows can use cmake -P cmake/bootstrap-ocr.cmake")
  endif()
  add_library(dp_ort SHARED IMPORTED)
  if(WIN32)
    set_target_properties(dp_ort PROPERTIES
      IMPORTED_LOCATION "${DP_ORT_ROOT}/lib/onnxruntime.dll"
      IMPORTED_IMPLIB "${DP_ORT_ROOT}/lib/onnxruntime.lib"
      INTERFACE_INCLUDE_DIRECTORIES "${DP_ORT_ROOT}/include")
  else()
    set_target_properties(dp_ort PROPERTIES IMPORTED_LOCATION "${DP_ORT_ROOT}/lib/libonnxruntime.dylib"
      INTERFACE_INCLUDE_DIRECTORIES "${DP_ORT_ROOT}/include")
  endif()
  add_library(dp_ocr src/ocr_onnx.cpp)
  target_link_libraries(dp_ocr PUBLIC dp_core PRIVATE dp_ort opencv_core opencv_imgproc)
  if(MSVC)
    target_compile_options(dp_ocr PRIVATE /W4)
  endif()
else()
  add_library(dp_ocr src/ocr_unavailable.cpp)
  target_link_libraries(dp_ocr PUBLIC dp_core)
endif()
